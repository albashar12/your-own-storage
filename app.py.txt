async def upload_to_telegram(file_path):
    client = TelegramClient('anon', API_ID, API_HASH)
    await client.start(bot_token=BOT_TOKEN)
    try:
        # force_document=True ব্যবহার করার কারণে টেলিগ্রাম ফাইল কম্প্রেস করবে না এবং অরিজিনাল কোয়ালিটি বজায় থাকবে
        await client.send_file(int(CHANNEL_ID), file_path, force_document=True)
        success = True
    except Exception as e:
        print(f"Upload error: {e}")
        success = False
    await client.disconnect()
    return success

@app.route('/')
def home():
    if 'user' in session:
        return render_template_string(HTML_TEMPLATE)
    return redirect(url_for('login'))

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')
        if username in users_db and users_db[username]["password"] == password:
            session['user'] = username
            return redirect(url_for('home'))
        return render_template_string(HTML_TEMPLATE, msg="Invalid username or password")
    return render_template_string(HTML_TEMPLATE)

@app.route('/register', methods=['GET', 'POST'])
def register():
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')
        question = request.form.get('question')
        answer = request.form.get('answer')
        
        if username == 'admin' or username in users_db:
            return render_template_string(HTML_TEMPLATE, msg="Username not available or already exists!")
        
        users_db[username] = {
            "password": password,
            "question": question,
            "answer": answer
        }
        session['user'] = username
        return redirect(url_for('home'))
    return render_template_string(HTML_TEMPLATE)

@app.route('/forgot', methods=['GET', 'POST'])
def forgot():
    if request.method == 'POST':
        username = request.form.get('username')
        answer = request.form.get('answer')
        new_password = request.form.get('new_password')
        
        if username in users_db and users_db[username]["answer"].strip().lower() == answer.strip().lower():
            users_db[username]["password"] = new_password
            return render_template_string(HTML_TEMPLATE, msg="Password reset successfully! Please login.")
        return render_template_string(HTML_TEMPLATE, msg="Invalid username or security answer!")
    return render_template_string(HTML_TEMPLATE)

@app.route('/upload', methods=['POST'])
def upload():
    if 'user' not in session or session['user'] == 'admin':
        return redirect(url_for('login'))
    
    if 'file' not in request.files:
        return redirect(url_for('home'))
    
    file = request.files['file']
    if file.filename == '':
        return redirect(url_for('home'))
    
    if file:
        filename = secure_filename(file.filename)
        os.makedirs("temp", exist_ok=True)
        file_path = os.path.join("temp", filename)
        file.save(file_path)
        
        # Async upload execution
        loop = asyncio.new_event_loop()
        asyncio.set_event_loop(loop)
        success = loop.run_until_complete(upload_to_telegram(file_path))
        
        # Remove temporary local file
        if os.path.exists(file_path):
            os.remove(file_path)
            
        if success:
            return render_template_string(HTML_TEMPLATE, msg="File uploaded successfully to Telegram!")
        else:
            return render_template_string(HTML_TEMPLATE, msg="File upload failed! Check Telegram credentials.")
            
    return redirect(url_for('home'))

@app.route('/logout')
def logout():
    session.pop('user', None)
    return redirect(url_for('login'))

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)
